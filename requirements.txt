# EDLD (ED Live Dashboard) requirements.txt
#
# IMPORTANT: Not all dependencies should be installed via pip.
# Run install.sh instead of pip install -r requirements.txt directly.
#
# ── System package manager (NOT pip) ─────────────────────────────────────────
#
#   psutil has C extensions that require system libraries.  Install it via your
#   distro package manager:
#
#   psutil (required for session management):
#     Arch:          sudo pacman -S python-psutil
#     Debian/Ubuntu: sudo apt install python3-psutil
#     Fedora:        sudo dnf install python3-psutil
#
# ── pip ──────────────────────────────────────────────────────────────────────
#
discord-webhook>=1.3.0
#
# Radio tab (Crew / Alerts window).  MIT; decodes MP3, Ogg Vorbis and FLAC and
# plays through the system's audio output, with no player to install.  EDLD
# runs without it — the Radio tab says it is missing.
miniaudio>=1.71
#
# Server mode (-s), for the EDAM mobile client.  Only needed if you use it;
# EDLD says so if you turn the server on without them.  See docs/SERVER.md.
#   cryptography — the server's key and certificate.  Apache 2.0 / BSD.
#                  Arch: pacman -S python-cryptography
#   segno        — draws the pairing QR code.  BSD.  Without it, --pair still
#                  prints the code and the link.
cryptography>=42.0
segno>=1.5
#
# ── UI frontends ─────────────────────────────────────────────────────────────
#
# Both are optional and independent: install whichever interfaces you want.
# EDLD tells you which one is missing if you ask for an interface you have not
# installed, and the other modes keep working.
#
# Textual TUI (--tui, the default terminal dashboard):
textual>=0.47.0
# Rich arrives with Textual, but the terminal blocks import it directly, so it
# is declared rather than assumed.  13.3.3 is the floor Textual 0.47 requires.
rich>=13.3.3
#
# PySide6 desktop window (--gui).  Qt for Python, LGPL v3 — see
# docs/LICENSING.md.  Comment this out if you only ever use the terminal
# interfaces; it is by far the largest dependency.
PySide6>=6.6.0
#
# ── stdlib (no install needed) ────────────────────────────────────────────────
#   tomllib, argparse, json, pathlib, re, queue, threading, datetime
